file(GLOB EXAMPLE_ENTRIES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/*")
foreach(EXAMPLE_DIR IN LISTS EXAMPLE_ENTRIES)
    if(IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/${EXAMPLE_DIR}" AND
       EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${EXAMPLE_DIR}/CMakeLists.txt")
        add_subdirectory("${EXAMPLE_DIR}")
    endif()
endforeach()
